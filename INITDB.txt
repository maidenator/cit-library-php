-- 1. DATABASE INITIALIZATION
CREATE DATABASE IF NOT EXISTS citu_library;
USE citu_library;

-- 2. TABLE STRUCTURES
CREATE TABLE BOOK (
    BookID VARCHAR(10) PRIMARY KEY,
    Title VARCHAR(255) NOT NULL,
    Author VARCHAR(255) NOT NULL,
    ISBN VARCHAR(20) NOT NULL,
    PublishedYear INT,
    AvailabilityStatus ENUM('Available', 'Checked Out') DEFAULT 'Available',
    BorrowCount INT DEFAULT 0
);

CREATE TABLE USER (
    UserID VARCHAR(20) PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    Password VARCHAR(255) NOT NULL,
    Role ENUM('Student', 'Faculty', 'Admin') DEFAULT 'Student',
    AccountStatus ENUM('Active', 'Restricted') DEFAULT 'Active',
    MaxLoanLimit INT DEFAULT 5
);

CREATE TABLE LOAN (
    LoanID INT AUTO_INCREMENT PRIMARY KEY,
    UserID VARCHAR(20),
    BookID VARCHAR(10),
    BorrowDate DATE,
    DueDate DATE,
    ReturnDate DATE NULL,
    LoanStatus ENUM('Active', 'Returned') DEFAULT 'Active',
    FOREIGN KEY (UserID) REFERENCES USER(UserID),
    FOREIGN KEY (BookID) REFERENCES BOOK(BookID)
);

CREATE TABLE RESERVATION (
    ReservationID INT AUTO_INCREMENT PRIMARY KEY,
    UserID VARCHAR(20),
    BookID VARCHAR(10),
    RequestDate DATE,
    QueueStatus ENUM('Pending', 'Fulfilled', 'Cancelled') DEFAULT 'Pending',
    FOREIGN KEY (UserID) REFERENCES USER(UserID),
    FOREIGN KEY (BookID) REFERENCES BOOK(BookID)
);
